@startuml
actor User
participant "MealDetailsFragment" as Fragment
participant "MealDetailsPresenter" as Presenter
participant "SaludRepository" as Repository
database "RoomLocalDB" as DB

User -> Fragment: click btnFavorite
Fragment -> Presenter: toggleFavorite(meal)
alt meal is not favorite
    Presenter -> Repository: addMealToFavourites(meal)
    Repository -> DB: insert meal into favorites
else meal is favorite
    Presenter -> Repository: removeMealFromFavourites(meal)
    Repository -> DB: delete meal from favorites
end
Presenter -> Fragment: updateFavoriteButton(newStatus)
@enduml
